      SUBROUTINE Doppler_Header( line )
C
C process doppler header (in line):
C
C day0, month0, year0, hour0, min0 - from header
C MJD_dop, UTC_dop, t_dop = UTC_dop + MJD_dop - calculated
C delta_t, f_corr, f_deriv, f_recv, f_res, fi_res, s_max - from line
C   after header
C dd_t0 = 0.04 from 2 lines after header
C
C dop_error = 1 -- next 'line' - header (?)
C dop_error = 2 -- after next 'line' - header (?)
C
      character(250) line
C
      Integer, parameter:: N_dop = 25
C
      Integer i, j, n, nn
      Real(8) sec0, JD_dop
C
      Integer day0,month0,year0, hour0,min0
      Integer dop_error
C
      Real(8) delta_t, f_corr, f_deriv, f_recv, f_res, fi_res, s_max
      Real(8) dd_t0, t_dop, MJD_dop, UTC_dop, prev_dt
	character(6) no1, no2
C
      COMMON /DOPPLER/ delta_t, f_corr, f_deriv, f_recv, f_res, fi_res
     +               , s_max
     +               , dd_t0, t_dop, MJD_dop, UTC_dop, prev_dt
     +               , day0, month0, year0, hour0, min0
     +               , dop_error
C
      if (line(1:1) /= "%" .or. line(2:2) /= "H")
     +   stop ' Invalid %H in Header line of doppler residuals file'
      nn = len_trim(line)
      n = scan(line(1 : nn), " ")
      if (n == 0) stop ' Invalid Header line of doppler residuals file'
      do while (line(n:n) == " ")
          n = n + 1
      enddo
      i = n
      do while (i <= nn)
          j = scan(line(i : nn), "-:")
          if (j == 0) then
              i = nn + 1
          else
              i = i + j
              line(i-1:i-1) = " "
          endif
      enddo
      READ (line(n:nn),*,IOSTAT=ios) day0,month0,year0,hour0,min0,sec0
      if (ios/= 0) stop ' Error reading date/time from doppler file'
C
c      write(*,*) day0,month0,year0, hour0,min0,sec0
c      pause
C  Read the second and third lines of dopler file to
C  calculate the time interval between phase data
      READ (21,'(A250)',IOSTAT=ios) line
      if (ios/= 0)
     +  stop ' Error reading next line after header from doppler file'
      if (line(1:1) == "%" .and. line(2:2) == "H") then
          dop_error = 1
          return
      endif
cc      READ (line,*,IOSTAT=ios)
cc     +      delta_t, f_corr, f_deriv, f_recv, f_res, fi_res, s_max !f=8.4
      READ (line,*,IOSTAT=ios)
     +      delta_t, no1, no2, f_recv, f_res, fi_res, s_max  !f=15
c	write (*,*) delta_t, no1, no2, f_recv, f_res, fi_res, s_max
c	pause

      if (ios/= 0)
     +  stop ' Invalid format of next line after header in doppler file'
      READ (21,'(A250)',IOSTAT=ios) line
      if (ios/= 0)
     + stop ' Error reading 2 next lines after header from doppler file'
      if (line(1:1) == "%" .and. line(2:2) == "H") then
          dop_error = 2
          return
      endif
      READ (line,*,IOSTAT=ios) dd_t0
      if (ios/= 0)
     +  stop ' Error reading two lines after header from doppler file'
C  Step (=0.04 s)
      dd_t0 = dfloat(int(dd_t0*1.d3)-int(delta_t*1.d3))*1.d-3  ! 0.04 s
C  Number of phases during one segment is
C  jj = dt * 25  (step = 0.04 s) (dt is N_wr_tot in seconds)
C           jj = N_wr_tot * int(1.d0/dd_t0)
c	 write(*,*) jj, t1, t2
c	 write(*,*) jj, dd_t0
C  Test for future
      if ( int(1.d0/dd_t0) /= N_dop ) then
       stop 'STOP in INTEGR6. N_dop has to be 1/dt (dt is step in file
     * of dopler);if dt=0.04s N_dop=25.Change N_dop in INTEGR6'
      endif
C  Back one record
      backspace(21)
C  Calculation of the beginning of the phase file
      call JULIAN(year0,month0,day0, JD_dop)
      MJD_dop = JD_dop - 2400000.5d0
      UTC_dop =
     =     (dfloat(hour0)+dfloat(min0)/60.d0+sec0/3600.d0)/24.d0
C  MJD of beginning of dopler file
      t_dop =  UTC_dop + MJD_dop
C
      dop_error = 0
      return
      END SUBROUTINE Doppler_Header

      SUBROUTINE Doppler_Times
C
C print times from doppler file (in case of error)
C
      Integer day0,month0,year0, hour0,min0
      Integer dop_error
C
      Integer i
      Real(8) tt
C
      Real(8) CDEGRAD, CARCRAD, CTIMRAD, SECDAY, JD2000, JUL_CENT
      Real(8) delta_t, f_corr, f_deriv, f_recv, f_res, fi_res, s_max
      Real(8) dd_t0, t_dop, MJD_dop, UTC_dop, prev_dt
C
      COMMON /MATH2/ CDEGRAD, CARCRAD, CTIMRAD, SECDAY, JD2000, JUL_CENT
      COMMON /DOPPLER/ delta_t, f_corr, f_deriv, f_recv, f_res, fi_res
     +               , s_max
     +               , dd_t0, t_dop, MJD_dop, UTC_dop, prev_dt
     +               , day0, month0, year0, hour0, min0
     +               , dop_error
C
      write(*,*) 'MJD H M S DD MM YYYY of doppler file:',
     +           MJD_dop, hour0, min0, sec0, day0, month0, year0
      write(*,*) 'Offset of previous doppler record =',prev_dt,'s'
      tt = UTC_dop * secday + prev_dt
      i = int(tt / secday)
      MJD_dop = MJD_dop + i
      tt = tt - i * secday
      hour0 = int(tt / 3600.)
      tt = tt - hour0 * 3600
      min0 = int(tt / 60.)
      tt = tt - min0 * 60
      write(*,*) 'MJD H M S of previous doppler record:',
     +           MJD_dop, hour0, min0, tt
      return
      END SUBROUTINE Doppler_Times
